    47→    initConnectivityMonitoring() {
    48→        const updateStatus = async () => {
    49→            const el = document.getElementById('internetStatus');
    50→            if (!el) return;
    51→
    52→            // First check navigator.onLine for a quick hint
    53→            let isOnline = navigator.onLine;
    54→
    55→            // If it says online, double check with a real fetch
    56→            if (isOnline) {
    57→                try {
    58→                    // Try to fetch a small resource to confirm actual internet access
    59→                    // Using no-cors and a cache-busting timestamp
    60→                    await fetch('https://www.google.com/favicon.ico?' + Date.now(), {
    61→                        mode: 'no-cors',
    62→                        method: 'HEAD',
    63→                        cache: 'no-store'
    64→                    });
    65→                    isOnline = true;
    66→                } catch (e) {
    67→                    // If fetch fails, we are likely offline or internet is blocked
    68→                    isOnline = false;
    69→                }
    70→            }
    71→
    72→            el.className = `gdrive-status ${isOnline ? 'connected' : 'error'}`;
    73→            el.innerHTML = `
    74→                <i class="fas fa-circle"></i>
    75→                <span>Internet: ${isOnline ? 'Online' : 'Offline'}</span>
    76→            `;
    77→        };
    78→
    79→        window.addEventListener('online', updateStatus);
    80→        window.addEventListener('offline', updateStatus);
    81→
    82→        // Initial check
    83→        updateStatus();
    84→
    85→        // Periodic check every 15 seconds to catch ISP drops or router issues
    86→        setInterval(updateStatus, 15000);
    87→    }

   412→        if (disconnectBtn) disconnectBtn.style.display = connected ? 'inline-flex' : 'none';

   414→        if (userInfo) userInfo.style.display = connected ? 'block' : 'none';
   415→
   416→        // Update status badge
   417→        if (statusContainer) {
   418→            let statusText = connected ? 'Connected' : 'Not Connected';
   419→            let statusClass = connected ? 'connected' : 'disconnected';
   420→
   421→            if (status.uploading) {
   422→                statusText = 'Uploading...';
   423→                statusClass = 'uploading';
   424→            }
   425→
   426→            statusContainer.innerHTML = `
   427→                <div class="gdrive-status ${statusClass}">
   428→                    <i class="fas fa-circle"></i>
   429→                    <span>${statusText}</span>
   430→                </div>
   431→            `;
   432→        }
   433→
   434→        // Update last backup time
   435→        const lastBackupEl = document.getElementById('gdriveLastBackup');
   436→        if (lastBackupEl) {
   437→            if (status.lastBackup) {
   438→                const date = new Date(status.lastBackup);

   543→    addToCart(product) {
   544→        const existing = this.cart.find(item => item.id === product.id);
   545→        if (existing) {
   546→            existing.quantity++;
   547→            existing.total = existing.quantity * existing.price;
   548→        } else {
   549→            this.cart.push({
   550→                ...product,
   551→                quantity: 1,
   552→                total: Number(product.price) || 0,
   553→                name: product.name,
   554→                price: Number(product.price) || 0
   555→            });
   556→        }
   557→        this.ui.renderCart();
   558→        this.showToast('Added to cart');
   559→
   560→        // Show cart sidebar
   561→        const sidebar = document.getElementById('cartSidebar');

   586→    async saveProduct(e) {
   587→        e.preventDefault();
   588→        const id = this.editingProductId;
   589→        const name = document.getElementById('productName').value;
   590→        const categoryId = document.getElementById('productCategory').value;
   591→
   592→        // Find which quantity type is selected
   593→        const qtyRadio = document.querySelector('input[name="productQuantity"]:checked');
   594→        const qtyType = qtyRadio ? qtyRadio.value : 'pcs';
   595→
   596→        // Get the friendly display text from the data attribute (e.g. "1 Kg")
   597→        let quantityType = qtyRadio ? qtyRadio.dataset.display : 'Pieces';
   598→
   599→        // If Custom is selected, use the manual text input
   600→        if (qtyType === 'custom') {
   601→            const customText = document.getElementById('customQuantityText').value;
   602→            quantityType = customText || 'Custom';
   603→        }

   861→        if (this.cart.length === 0) return;
   862→

   864→

   867→        let nextIdNum = 1;

   870→            nextIdNum = (lastOrder.id || 0) + 1;

   878→            timestamp: Date.now(),

   880→            total: total,
   881→            status: 'completed',

   885→        this.ui.showReceipt(this.pendingOrder);

   899→    async printReceipt() {
   900→        if (!this.pendingOrder) return;
   901→
   902→        // Remember if we were in fullscreen
   903→        const wasFullscreen = !!document.fullscreenElement;
   904→
   905→        try {
   906→            // 1. Save order to database and clear cart
   907→            await this.dataService.saveOrder(this.pendingOrder);
   908→            this.cart = [];
   909→            this.ui.renderCart();
   910→
   911→            // 2. Perform Print Sequence (Dual separate jobs as requested)
   912→            // First Print: Customer Copy
   913→            const receiptContent = document.getElementById('receiptContent');
   914→            if (receiptContent) {
   915→                receiptContent.innerHTML = this.ui.renderReceiptHTML(this.pendingOrder, 'CUSTOMER COPY');
   916→            }
   917→            window.print();
   918→
   919→            // Small delay for browser to process the first job before second
   920→            await new Promise(r => setTimeout(r, 1000));
   921→
   922→            // Second Print: Counter Copy
   923→            if (receiptContent) {
   924→                receiptContent.innerHTML = this.ui.renderReceiptHTML(this.pendingOrder, 'COUNTER COPY');
   925→                window.print();
   926→            }
   927→
   928→            this.ui.showToast("Order Saved and Printed", "success");
   929→            this.closeReceiptModal();
   930→
   931→            // Refresh Sales view if the tab is active or just to keep state synced
   932→            if (this.ui.renderSales) this.ui.renderSales();
   933→
   934→            // Try to restore fullscreen if it was lost during printing
   935→            if (wasFullscreen && !document.fullscreenElement) {
   936→                setTimeout(() => {
   937→                    document.documentElement.requestFullscreen().catch(() => {
   938→                        // Silent fail if browser blocks gesture-less restoration
   939→                    });
   940→                }, 1000);
   941→            }
   942→        } catch (error) {
   943→            console.error("Print failed:", error);
   944→            this.ui.showToast("Failed to save order", "error");
   945→        }
   946→    }
